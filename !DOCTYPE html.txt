  const { createApp, ref, computed, onMounted, nextTick } = Vue;

  createApp({
    setup() {
      const currentTab = ref('schedule');
      const tabs = [
        { id: 'schedule', icon: 'map-pin' },
        { id: 'info', icon: 'info' },
        { id: 'shopping', icon: 'shopping-bag' },
        { id: 'expenses', icon: 'dollar-sign' }
      ];

      // 天氣資訊預設
      const currentWeather = ref({
        icon: 'sun',
        temp: 16,
        feelsLike: 15,
        condition: '晴朗舒適'
      });

      // 日期資料
      const selectedDayIndex = ref(0);
      const days = [
        { week: 'Fri', date: '28', title: 'DAY 1 海雲台散步、膠囊列車' },
        { week: 'Sat', date: '29', title: 'DAY 2 慶州古都之旅' },
        { week: 'Sun', date: '30', title: 'DAY 3 廣安里海景行程' }
      ];
      const currentDay = computed(() => days[selectedDayIndex.value]);

      // 行程分類與資料
      const filterCategory = ref('全部');
      const schedules = ref([
        // DAY 1
        { id: 1, day: 0, category: '航班', title: '去程航班 LJ578', flightNo: 'LJ578', depCode: 'TPE', depTime: '02:40', arrCode: 'PUS', arrTime: '06:10', naverUrl: 'https://map.naver.com' },
        { id: 2, day: 0, category: '景點', title: '海雲台藍線公園 (膠囊列車)', time: '09:30 - 11:30', ticket: '₩35,000 (雙人)', note: '從尾浦搭到青沙浦，拍攝經典海景列車', travelTime: '25分鐘', transportIcon: 'bus', naverUrl: 'https://map.naver.com/v5/search/%ED%95%B4%EC%9A%B4%EB%8C%80%20%EB%B8%94%EB%A3%A8%EB%9D%BC%EC%9D%B8%ED%8C%8C%ED%81%AC' },
        { id: 3, day: 0, category: '美食', title: '極東豬肉湯飯 (海雲台店)', time: '12:00 - 13:00', ticket: '-', note: '釜山必吃名店，湯頭濃郁美味', travelTime: '10分鐘', transportIcon: 'footprints', naverUrl: 'https://map.naver.com' },
        // DAY 2
        { id: 4, day: 1, category: '景點', title: '慶州瞻星台 & 皇理團路', time: '10:00 - 16:00', ticket: '免費', note: '體驗古都歷史風情，逛文青特色小店', travelTime: '60分鐘', transportIcon: 'car', naverUrl: 'https://map.naver.com' },
        // DAY 3
        { id: 5, day: 2, category: '景點', title: '廣安里海水浴場', time: '14:00 - 17:00', ticket: '免費', note: '欣賞廣安大橋海景與沿岸咖啡廳', travelTime: '30分鐘', transportIcon: 'bus', naverUrl: 'https://map.naver.com' }
      ]);

      const filteredSchedules = computed(() => {
        return schedules.value.filter(s => s.day === selectedDayIndex.value && (filterCategory.value === '全部' || s.category === filterCategory.value));
      });

      // 匯率計算
      const krwInput = ref(50000);

      // 購物清單
      const newShopItem = ref('');
      const shoppingList = ref([
        { name: 'Olive Young 護膚品', done: false },
        { name: '釜山限定魚板禮盒', done: true }
      ]);
      const addShoppingItem = () => {
        if (newShopItem.value.trim()) {
          shoppingList.value.push({ name: newShopItem.value.trim(), done: false });
          newShopItem.value = '';
        }
      };

      // 花費紀錄
      const expenses = ref([
        { category: '飲食', title: '海雲台海鮮煎餅', date: '2024-11-28', amount: 25000, payment: '現金' },
        { category: '交通', title: '膠囊列車車票', date: '2024-11-28', amount: 35000, payment: '信用卡' }
      ]);
      const totalExpenses = computed(() => expenses.value.reduce((sum, item) => sum + Number(item.amount), 0));

      // Modal & 表單控制
      const showModal = ref(false);
      const isEditing = ref(false);
      const editingId = ref(null);
      const formSchedule = ref({ category: '景點', title: '', time: '', ticket: '', note: '' });
      const formExpense = ref({ category: '飲食', title: '', amount: '', payment: '現金' });

      const openAddModal = () => {
        isEditing.value = false;
        formSchedule.value = { category: '景點', title: '', time: '', ticket: '', note: '' };
        formExpense.value = { category: '飲食', title: '', amount: '', payment: '現金' };
        showModal.value = true;
      };

      const openEditModal = (item) => {
        isEditing.value = true;
        editingId.value = item.id;
        formSchedule.value = { ...item };
        showModal.value = true;
      };

      const saveForm = () => {
        if (currentTab.value === 'schedule') {
          if (isEditing.value) {
            const idx = schedules.value.findIndex(s => s.id === editingId.value);
            if (idx !== -1) schedules.value[idx] = { ...schedules.value[idx], ...formSchedule.value };
          } else {
            schedules.value.push({
              id: Date.now(),
              day: selectedDayIndex.value,
              ...formSchedule.value,
              naverUrl: 'https://map.naver.com'
            });
          }
        } else if (currentTab.value === 'expenses') {
          expenses.value.push({
            ...formExpense.value,
            date: new Date().toISOString().split('T')[0]
          });
        }
        showModal.value = false;
        nextTick(() => lucide.createIcons());
      };

      const confirmDelete = () => {
        if (confirm('確定要刪除這個行程項目嗎？')) {
          schedules.value = schedules.value.filter(s => s.id !== editingId.value);
          showModal.value = false;
        }
      };

      const getCategoryBadge = (cat) => {
        const map = {
          '景點': 'bg-emerald-100 text-emerald-700',
          '美食': 'bg-orange-100 text-orange-700',
          '住宿': 'bg-purple-100 text-purple-700',
          '其他': 'bg-slate-100 text-slate-700'
        };
        return map[cat] || 'bg-slate-100 text-slate-700';
      };

      onMounted(() => {
        lucide.createIcons();

        // 初始化 Sortable 拖曳排序
        const el = document.getElementById('schedule-list');
        if (el) {
          Sortable.create(el, {
            handle: '.handle',
            animation: 150,
            onEnd: () => {
              // 可在此獲取新的拖曳順序
            }
          });
        }
      });

      return {
        currentTab, tabs, currentWeather, selectedDayIndex, days, currentDay,
        filterCategory, schedules, filteredSchedules, krwInput, newShopItem,
        shoppingList, addShoppingItem, expenses, totalExpenses, showModal,
        isEditing, formSchedule, formExpense, openAddModal, openEditModal,
        saveForm, confirmDelete, getCategoryBadge
      };
    }
  }).mount('#app');
</script>
</body>
</html>
